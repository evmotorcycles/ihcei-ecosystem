# Python dependencies for the full reproduction run (bash reproduce_all.sh).
#
# Most suites in this repository are deliberately pure-stdlib and need none of
# these. The list exists because a fresh container without them fails SEVEN
# suites with bare ModuleNotFoundError, which reads like a broken repository
# rather than a missing install. reproduce_all.sh now checks for them up front
# and names what is missing instead.
#
#   python3 -m pip install -r requirements.txt
#
# Versions below are the ones the suites were last verified green against
# (66/66 on 2026-08-09). They are recorded for provenance, not pinned, so a
# newer release that breaks a result surfaces as a failure rather than being
# silently avoided.

numpy          # >= 2.4.6   yeast/GitHub cohort maths, physics-agency
scipy          # >= 1.17.1  statistics used by the LISM cohorts
pandas         # >= 3.0.5   cohort tables
statsmodels    # >= 0.14.6  VIF and regression fits (cohort-audit gap closure)
networkx       # >= 3.6.1   physics-agency 3D coordinate emergence
scikit-learn   # >= 1.9.0   GitHub cohort attestation in lism-cohorts
jax            # >= 0.10.2  financial-system sovereign mesh telemetry
openpyxl       # >= 3.1.5   reads the frozen .xlsx datasets in financial-system
pytest         # >= 9.1.1   test runner
